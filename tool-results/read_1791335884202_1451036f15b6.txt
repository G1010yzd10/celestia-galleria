   469→   469→  function renderReflection(
   470→   470→    renderer: THREE.WebGLRenderer,
   471→   471→    scene: THREE.Scene,
   472→   472→    camera: THREE.PerspectiveCamera,
   473→   473→    hidden: THREE.Object3D[]
   474→   474→  ) {
   475→   475→    camera.getWorldPosition(_camPos);
   476→   476→    camera.getWorldDirection(_camDir);
   477→   477→    _camUp.set(0, 1, 0).applyQuaternion(camera.getWorldQuaternion(new THREE.Quaternion()));
   478→   478→
   479→   479→    // mirror camera across the y = FLOOR_Y plane
   480→   480→    _mPos.set(_camPos.x, 2 * FLOOR_Y - _camPos.y, _camPos.z);
   481→   481→    const mDir = new THREE.Vector3(_camDir.x, -_camDir.y, _camDir.z);
   482→   482→    const mUp = new THREE.Vector3(_camUp.x, -_camUp.y, _camUp.z);
   483→   483→    _mTarget.copy(_mPos).add(mDir);
   484→   484→    virtualCam.up.copy(mUp);
   485→   485→    virtualCam.position.copy(_mPos);
   486→   486→    virtualCam.lookAt(_mTarget);
   487→   487→    virtualCam.near = camera.near;
   488→   488→    virtualCam.far = camera.far;
   489→   489→    virtualCam.projectionMatrix.copy(camera.projectionMatrix);
   490→   490→    virtualCam.updateMatrixWorld();
   491→   491→
   492→   492→    // projective sampling matrix (world → mirror-clip → [0,1])
   493→   493→    texMat.copy(BIAS)
   494→   494→      .multiply(virtualCam.projectionMatrix)
   495→   495→      .multiply(virtualCam.matrixWorldInverse);
   496→   496→    (fu.uTexMat.value as THREE.Matrix4).copy(texMat);
   497→   497→
   498→   498→    // oblique near-plane clipping so nothing below the floor leaks in
   499→   499→    _plane.setFromNormalAndCoplanarPoint(
   500→   500→      new THREE.Vector3(0, 1, 0),
   501→   501→      new THREE.Vector3(0, FLOOR_Y, 0)
   502→   502→    );
   503→   503→    _plane.applyMatrix4(virtualCam.matrixWorldInverse);
   504→   504→    const clip = new THREE.Vector4(
   505→   505→      _plane.normal.x, _plane.normal.y, _plane.normal.z, _plane.constant
   506→   506→    );
   507→   507→    const pm = virtualCam.projectionMatrix;
   508→   508→    _q.x = (Math.sign(clip.x) + pm.elements[8]) / pm.elements[0];
   509→   509→    _q.y = (Math.sign(clip.y) + pm.elements[9]) / pm.elements[5];
   510→   510→    _q.z = -1.0;
   511→   511→    _q.w = (1.0 + pm.elements[10]) / pm.elements[14];
   512→   512→    clip.multiplyScalar(2.0 / clip.dot(_q));
   513→   513→    pm.elements[2] = clip.x;
   514→   514→    pm.elements[6] = clip.y;
   515→   515→    pm.elements[10] = clip.z + 1.0;
   516→   516→    pm.elements[14] = clip.w;
   517→   517→    virtualCam.projectionMatrixInverse.copy(pm).invert();
   518→   518→
   519→   519→    // render the reflected world (self + caller-hidden objects removed)
   520→   520→    for (const f of floorSegs) f.visible = false;
   521→   521→    for (const h of hidden) h.visible = false;
   522→   522→    const prevRT = renderer.getRenderTarget();
   523→   523→    try {
   524→   524→      renderer.setRenderTarget(mirrorRT);
   525→   525→      renderer.render(scene, virtualCam);
   526→   526→    } finally {
   527→   527→      renderer.setRenderTarget(prevRT);
   528→   528→      for (const f of floorSegs) f.visible = true;
   529→   529→      for (const h of hidden) h.visible = true;
   530→   530→    }
   531→   531→  }
